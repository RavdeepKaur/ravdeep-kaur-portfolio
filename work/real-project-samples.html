<?php
// db.php
// Creates one reusable PDO connection for the application.
// ERRMODE_EXCEPTION makes database errors easier to catch and debug.
$dsn = "mysql:host=localhost;dbname=training_app;charset=utf8mb4";
$pdo = new PDO($dsn, "app_user", "strong_password", [
  PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
  PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC
]);

// CREATE: insert a new customer record with named placeholders.
function createCustomer(PDO $pdo, string $name, string $email): bool {
  $sql = "INSERT INTO customers (name, email, status)
          VALUES (:name, :email, 'Active')";

  $stmt = $pdo->prepare($sql);
  return $stmt->execute([
    ":name" => trim($name),
    ":email" => trim($email)
  ]);
}

// READ: retrieve all customers for the admin table.
function getCustomers(PDO $pdo): array {
  $stmt = $pdo->query(
    "SELECT customer_id, name, email, status
     FROM customers
     ORDER BY customer_id DESC"
  );

  return $stmt->fetchAll();
}

// UPDATE: edit a customer by primary key.
function updateCustomer(PDO $pdo, int $id, string $name, string $email): bool {
  $sql = "UPDATE customers
          SET name = :name, email = :email
          WHERE customer_id = :id";

  $stmt = $pdo->prepare($sql);
  return $stmt->execute([
    ":id" => $id,
    ":name" => trim($name),
    ":email" => trim($email)
  ]);
}

// DELETE: remove one customer by primary key.
function deleteCustomer(PDO $pdo, int $id): bool {
  $stmt = $pdo->prepare(
    "DELETE FROM customers WHERE customer_id = :id"
  );

  return $stmt->execute([":id" => $id]);
}

using System;
using System.Data;
using System.Data.SqlClient;

public partial class CustomerDashboard : Form
{
    private readonly string connectionString =
        @"Data Source=(LocalDB)\MSSQLLocalDB;
          AttachDbFilename=|DataDirectory|\CustomerManager.mdf;
          Integrated Security=True";

    public CustomerDashboard()
    {
        InitializeComponent();
        ConfigureGrid();
        LoadCustomers();
    }

    // Adds an Edit button column so each record can open the update workflow.
    private void ConfigureGrid()
    {
        DataGridViewButtonColumn editColumn = new DataGridViewButtonColumn();
        editColumn.Name = "EditAction";
        editColumn.HeaderText = "Action";
        editColumn.Text = "Edit";
        editColumn.UseColumnTextForButtonValue = true;
        customersGrid.Columns.Add(editColumn);
    }

    // Centralized read operation: retrieves records and binds them to DataGridView.
    private void LoadCustomers()
    {
        using SqlConnection connection = new SqlConnection(connectionString);
        using SqlDataAdapter adapter = new SqlDataAdapter(
            "SELECT CustomerID, FullName, City, Status, LastUpdated FROM Customers",
            connection
        );

        DataTable table = new DataTable();
        adapter.Fill(table);
        customersGrid.DataSource = table;
    }

    // Validation keeps incomplete data from reaching the database layer.
    private bool ValidateCustomerForm()
    {
        return !string.IsNullOrWhiteSpace(nameTextBox.Text) &&
               !string.IsNullOrWhiteSpace(cityTextBox.Text) &&
               statusComboBox.SelectedItem != null;
    }

    // CREATE operation using a parameterized query to avoid SQL injection.
    private void saveButton_Click(object sender, EventArgs e)
    {
        if (!ValidateCustomerForm())
        {
            MessageBox.Show("Please complete all required fields.");
            return;
        }

        string sql = @"INSERT INTO Customers (FullName, City, Status, LastUpdated)
                       VALUES (@name, @city, @status, @updated)";

        try
        {
            using SqlConnection connection = new SqlConnection(connectionString);
            using SqlCommand command = new SqlCommand(sql, connection);

            command.Parameters.AddWithValue("@name", nameTextBox.Text.Trim());
            command.Parameters.AddWithValue("@city", cityTextBox.Text.Trim());
            command.Parameters.AddWithValue("@status", statusComboBox.Text);
            command.Parameters.AddWithValue("@updated", DateTime.Today);

            connection.Open();
            command.ExecuteNonQuery();
            LoadCustomers();
            MessageBox.Show("Customer saved successfully.");
        }
        catch (SqlException ex)
        {
            MessageBox.Show("Database error: " + ex.Message);
        }
    }

    // UPDATE operation: saves edits for the selected customer record.
    private void updateButton_Click(object sender, EventArgs e)
    {
        if (customersGrid.CurrentRow == null || !ValidateCustomerForm())
        {
            MessageBox.Show("Select a customer and complete all required fields.");
            return;
        }

        int customerId = Convert.ToInt32(
            customersGrid.CurrentRow.Cells["CustomerID"].Value
        );

        string sql = @"UPDATE Customers
                       SET FullName = @name,
                           City = @city,
                           Status = @status,
                           LastUpdated = @updated
                       WHERE CustomerID = @id";

        using SqlConnection connection = new SqlConnection(connectionString);
        using SqlCommand command = new SqlCommand(sql, connection);

        command.Parameters.AddWithValue("@id", customerId);
        command.Parameters.AddWithValue("@name", nameTextBox.Text.Trim());
        command.Parameters.AddWithValue("@city", cityTextBox.Text.Trim());
        command.Parameters.AddWithValue("@status", statusComboBox.Text);
        command.Parameters.AddWithValue("@updated", DateTime.Today);

        connection.Open();
        command.ExecuteNonQuery();
        LoadCustomers();
        MessageBox.Show("Customer updated successfully.");
    }
}

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.chrome.ChromeOptions;

import java.util.HashSet;
import java.util.List;
import java.util.Set;

public class LinkAuditTest {

    private static final String DRIVER_PATH =
            "C:\\SeleniumTools\\ChromeDriver\\chromedriver.exe";

    public static void main(String[] args) {
        WebDriver driver = setupDriver();

        try {
            driver.get("https://www.selenium.dev/");
            List<WebElement> links = driver.findElements(By.tagName("a"));
            printLinkReport(driver.getCurrentUrl(), links);
        } finally {
            driver.quit();
        }
    }

    // Configures ChromeDriver for a local Selenium WebDriver run.
    private static WebDriver setupDriver() {
        System.setProperty("webdriver.chrome.driver", DRIVER_PATH);

        ChromeOptions options = new ChromeOptions();
        options.addArguments("--remote-allow-origins=*");

        WebDriver driver = new ChromeDriver(options);
        driver.manage().window().maximize();
        return driver;
    }

    // Reviews page links and reports missing labels, missing URLs, and duplicates.
    private static void printLinkReport(String pageUrl, List<WebElement> links) {
        int emptyLabels = 0;
        int missingUrls = 0;
        int duplicateUrls = 0;
        Set<String> uniqueUrls = new HashSet<>();

        System.out.println("Page tested: " + pageUrl);
        System.out.println("Total links found: " + links.size());

        for (WebElement link : links) {
            String label = link.getText().trim();
            String href = link.getAttribute("href");

            if (label.isEmpty()) {
                emptyLabels++;
            }

            if (href == null || href.isBlank()) {
                missingUrls++;
                continue;
            }

            if (!uniqueUrls.add(href)) {
                duplicateUrls++;
            }
        }

        System.out.println("Valid link URLs: " + uniqueUrls.size());
        System.out.println("Empty or missing link labels: " + emptyLabels);
        System.out.println("Missing URLs: " + missingUrls);
        System.out.println("Duplicate URLs requiring review: " + duplicateUrls);
    }
}

Mobile App ProjectReact Native To-Do List App

This mobile app project was designed as a practical React Native exercise for demonstrating state management, text input, list rendering, touch events, conditional styling, and simple CRUD-style interaction in a mobile interface. 

import React, { useState } from "react";
import {
  SafeAreaView,
  View,
  Text,
  TextInput,
  TouchableOpacity,
  FlatList,
  StyleSheet
} from "react-native";

export default function TodoApp() {
  // Stores the current value typed into the task input.
  const [task, setTask] = useState("");

  // Stores the list of task objects displayed in the FlatList.
  // Each task has an id, visible text, and completion status.
  const [tasks, setTasks] = useState([
    { id: "1", text: "Review JavaScript notes", done: false },
    { id: "2", text: "Build React Native UI", done: false }
  ]);

  // Adds a new task only when the input contains meaningful text.
  // Date.now() creates a simple unique id for classroom/demo use.
  function addTask() {
    if (!task.trim()) return;
    setTasks([...tasks, { id: Date.now().toString(), text: task, done: false }]);
    setTask("");
  }

  // Toggles one task between active and completed.
  // map() returns a new array so React can re-render the changed state.
  function toggleTask(id) {
    setTasks(tasks.map(item =>
      item.id === id ? { ...item, done: !item.done } : item
    ));
  }

  // Removes one task from the list by filtering it out by id.
  function deleteTask(id) {
    setTasks(tasks.filter(item => item.id !== id));
  }

  return (
    <SafeAreaView style={styles.container}>
      <Text style={styles.title}>To-Do List</Text>

      <View style={styles.inputRow}>
        <TextInput
          value={task}
          onChangeText={setTask}
          placeholder="Add a new task"
          style={styles.input}
        />
        <TouchableOpacity onPress={addTask} style={styles.addButton}>
          <Text style={styles.addText}>+</Text>
        </TouchableOpacity>
      </View>

      <FlatList
        data={tasks}
        keyExtractor={item => item.id}
        // renderItem controls the layout and behavior for each task row.
        renderItem={({ item }) => (
          <View style={styles.taskRow}>
            <TouchableOpacity onPress={() => toggleTask(item.id)}>
              <Text style={item.done ? styles.doneTask : styles.taskText}>
                {item.text}
              </Text>
            </TouchableOpacity>
            <TouchableOpacity onPress={() => deleteTask(item.id)}>
              <Text style={styles.deleteText}>Delete</Text>
            </TouchableOpacity>
          </View>
        )}
      />
    </SafeAreaView>
  );
}

    assignButton.disabled = true;

static void quicksort(int[] arr, int left, int right) {
  if (left < right) {
    int pivotIndex = partition(arr, left, right);
    quicksort(arr, left, pivotIndex - 1);
    quicksort(arr, pivotIndex + 1, right);
  }
}
